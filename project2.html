Второй проект: API блога

Соберём API для блога на Flask и SQLite: с проверкой данных, тестами и простой страницей на JavaScript. Понадобятся знания из разделов про базы данных и REST API. Времени уйдёт около часа.

Что получится

Весь код поместится в одном файле app.py. Последний пример можно запустить прямо на этой странице.

0 из 7 шагов пройдено

Шаги проекта

Отмечайте пройденное: прогресс сохранится в браузере.

  1. Шаг 1. Создайте проект

    Новая папка и своё окружение, как в первом проекте.

    Терминал 
    mkdir blog-api
cd blog-api
python -m venv .venv
.venv\Scripts\activate
pip install flask pytest

    На macOS и Linux окружение включает команда source .venv/bin/activate.

  2. Шаг 2. Подключите базу данных

    Соединение открывается на время запроса и закрывается в конце. Таблица создаётся при первом запуске, если её ещё нет. Функция create_app позволит тестам использовать отдельную базу.

    app.py 
    import sqlite3

from flask import Flask, g, jsonify, request

SCHEMA = """
CREATE TABLE IF NOT EXISTS posts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    body TEXT NOT NULL,
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
)
"""

def create_app(database="blog.db"):
    app = Flask(__name__)

    def db():
        if "db" not in g:
            g.db = sqlite3.connect(database)
            g.db.row_factory = sqlite3.Row
            g.db.execute(SCHEMA)
        return g.db

    @app.teardown_appcontext
    def close_db(error):
        connection = g.pop("db", None)
        if connection is not None:
            connection.close()

    # маршруты из следующих шагов идут здесь

    return app

  3. Шаг 3. Добавьте проверку данных

    Функция возвращает либо чистые данные, либо текст ошибки. Вынесите её над create_app, тогда её легко проверить отдельно.

    app.py 
    def validate_post(data):
    if not isinstance(data, dict):
        return None, "Нужен JSON-объект"
    title = str(data.get("title", "")).strip()
    body = str(data.get("body", "")).strip()

        return None, "Заголовок обязателен, не длиннее 120 символов"
    if not body:
        return None, "Текст обязателен"
    return {"title": title, "body": body}, None

  4. Шаг 4. Сделайте список и создание записи

    Значения передаются через знаки ?. Так база сама защищает запрос от SQL-инъекций. Никогда не вставляйте данные пользователя в SQL через f-строку.

    app.py 
        @app.get("/posts")
    def list_posts():
        rows = db().execute("SELECT * FROM posts ORDER BY id DESC").fetchall()
        return jsonify([dict(row) for row in rows])

    @app.post("/posts")
    def create_post():
        post, error = validate_post(request.get_json(silent=True))
        if error:
            return {"error": error}, 400
        cursor = db().execute(
            "INSERT INTO posts (title, body) VALUES (?, ?)",
            (post["title"], post["body"]),
        )
        db().commit()
        row = db().execute("SELECT * FROM posts WHERE id = ?", (cursor.lastrowid,)).fetchone()
        return dict(row), 201

  5. Шаг 5. Сделайте чтение, изменение и удаление

    Если записи нет, возвращаем ошибку 404, а не падаем.

    app.py 
        @app.get("/posts/<int:post_id>")
    def get_post(post_id):
        row = db().execute("SELECT * FROM posts WHERE id = ?", (post_id,)).fetchone()
        if row is None:
            return {"error": "Запись не найдена"}, 404
        return dict(row)

    @app.put("/posts/<int:post_id>")
    def update_post(post_id):
        post, error = validate_post(request.get_json(silent=True))
        if error:
            return {"error": error}, 400
        cursor = db().execute(
            "UPDATE posts SET title = ?, body = ? WHERE id = ?",
            (post["title"], post["body"], post_id),
        )
        db().commit()
        if cursor.rowcount == 0:
            return {"error": "Запись не найдена"}, 404
        return get_post(post_id)

    @app.delete("/posts/<int:post_id>")
    def delete_post(post_id):
        cursor = db().execute("DELETE FROM posts WHERE id = ?", (post_id,))
        db().commit()
        if cursor.rowcount == 0:
            return {"error": "Запись не найдена"}, 404
        return "", 204

    Запуск на своём компьютере: flask --app "app:create_app()" run --debug.

  6. Шаг 6. Напишите тесты

    Запустите командой pytest. Фикстура tmp_path даёт временную папку, поэтому тесты не трогают настоящую базу.

    test_app.py 
    import pytest

from app import create_app

@pytest.fixture
def client(tmp_path):
    return create_app(tmp_path / "test.db").test_client()

def test_create_and_read(client):
    created = client.post("/posts", json={"title": "Привет", "body": "Текст"})
    assert created.status_code == 201
    post_id = created.json["id"]
    assert client.get(f"/posts/{post_id}").json["title"] == "Привет"

def test_empty_title_rejected(client):
    assert client.post("/posts", json={"title": " ", "body": "Текст"}).status_code == 400

def test_delete(client):
    post_id = client.post("/posts", json={"title": "А", "body": "Б"}).json["id"]
    assert client.delete(f"/posts/{post_id}").status_code == 204
    assert client.get(f"/posts/{post_id}").status_code == 404

  7. Шаг 7. Добавьте страницу

    Файл положите в папку static, он откроется по адресу /static/index.html. Текст записи выводится через textContent, поэтому вставить чужой скрипт через него нельзя.

    static/index.html 
    <!doctype html>
<html lang="ru">
<meta charset="utf-8">
<title>Блог</title>
<h1>Блог</h1>
<form id="form">
  <input name="title" placeholder="Заголовок" required>
  <textarea name="body" placeholder="Текст" required></textarea>
  <button>Опубликовать</button>
</form>
<div id="posts"></div>
<script>
  const posts = document.getElementById("posts");
  const form = document.getElementById("form");

  async function load() {
    const items = await (await fetch("/posts")).json();
    posts.replaceChildren(...items.map((post) => {
      const article = document.createElement("article");
      const title = document.createElement("h2");
      const body = document.createElement("p");
      title.textContent = post.title;
      body.textContent = post.body;
      article.append(title, body);
      return article;
    }));
  }

  form.addEventListener("submit", async (event) => {
    event.preventDefault();
    const response = await fetch("/posts", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(Object.fromEntries(new FormData(form))),
    });
    if (response.ok) {
      form.reset();
      load();
    }
  });

  load();
</script>
</html>

Весь код и запуск

Здесь весь файл app.py и короткая проверка в конце. База создаётся во временном файле.

app.py целиком 
import sqlite3
import tempfile
from pathlib import Path

from flask import Flask, g, jsonify, request

SCHEMA = """
CREATE TABLE IF NOT EXISTS posts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    body TEXT NOT NULL,
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
)
"""

def validate_post(data):
    if not isinstance(data, dict):
        return None, "Нужен JSON-объект"
    title = str(data.get("title", "")).strip()
    body = str(data.get("body", "")).strip()
    if not title or len(title) > 120:
        return None, "Заголовок обязателен, не длиннее 120 символов"
    if not body:
        return None, "Текст обязателен"
    return {"title": title, "body": body}, None

def create_app(database="blog.db"):
    app = Flask(__name__)

    def db():
        if "db" not in g:
            g.db = sqlite3.connect(database)
            g.db.row_factory = sqlite3.Row
            g.db.execute(SCHEMA)
        return g.db

    @app.teardown_appcontext
    def close_db(error):
        connection = g.pop("db", None)
        if connection is not None:
            connection.close()

    @app.get("/posts")
    def list_posts():
        rows = db().execute("SELECT * FROM posts ORDER BY id DESC").fetchall()
        return jsonify([dict(row) for row in rows])

    @app.post("/posts")
    def create_post():
        post, error = validate_post(request.get_json(silent=True))
        if error:
            return {"error": error}, 400
        cursor = db().execute(
            "INSERT INTO posts (title, body) VALUES (?, ?)",
            (post["title"], post["body"]),
        )
        db().commit()
        row = db().execute("SELECT * FROM posts WHERE id = ?", (cursor.lastrowid,)).fetchone()
        return dict(row), 201

    @app.get("/posts/<int:post_id>")
    def get_post(post_id):
        row = db().execute("SELECT * FROM posts WHERE id = ?", (post_id,)).fetchone()
        if row is None:
            return {"error": "Запись не найдена"}, 404
        return dict(row)

    @app.put("/posts/<int:post_id>")
    def update_post(post_id):
        post, error = validate_post(request.get_json(silent=True))
        if error:
            return {"error": error}, 400
        cursor = db().execute(
            "UPDATE posts SET title = ?, body = ? WHERE id = ?",
            (post["title"], post["body"], post_id),
        )
        db().commit()
        if cursor.rowcount == 0:
            return {"error": "Запись не найдена"}, 404
        return get_post(post_id)

    @app.delete("/posts/<int:post_id>")
    def delete_post(post_id):
        cursor = db().execute("DELETE FROM posts WHERE id = ?", (post_id,))
        db().commit()
        if cursor.rowcount == 0:
            return {"error": "Запись не найдена"}, 404
        return "", 204

    return app

# Проверка без настоящего сервера
with tempfile.TemporaryDirectory() as folder:
    client = create_app(Path(folder) / "blog.db").test_client()
    print("Пустой заголовок:", client.post("/posts", json={"title": "", "body": "x"}).status_code)
    created = client.post("/posts", json={"title": "Первая запись", "body": "Привет!"})
    print("Создана:", created.status_code, created.json["title"])
    post_id = created.json["id"]
    print("Изменена:", client.put(f"/posts/{post_id}", json={"title": "Новая", "body": "Текст"}).json["title"])
    print("Список:", [post["title"] for post in client.get("/posts").json])
    print("Удалена:", client.delete(f"/posts/{post_id}").status_code)
    print("Снова читаем:", client.get(f"/posts/{post_id}").status_code)

Что добавить самостоятельно

Эти идеи закрепят материал из других разделов.

  • Постраничный вывод: параметры ?limit=10&offset=20 и SQL LIMIT ? OFFSET ?.
  • Вход автора: создавать и удалять записи можно только после авторизации.
  • Поиск по заголовку через LIKE ?.
  • Публикация на PythonAnywhere: укажите в WSGI-файле application = create_app("/home/username/blog-api/blog.db").
